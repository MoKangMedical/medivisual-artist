<script>

    function calculate() {

        const B = parseFloat(document.getElementById('basePrice').value) || 0;

        

        // 获取各系数数量

        const q_basic = parseInt(document.getElementById('qty_basic').value) || 0;

        const q_high = parseInt(document.getElementById('qty_high').value) || 0;

        const q_immune = parseInt(document.getElementById('qty_immune').value) || 0;

        const q_render = parseInt(document.getElementById('qty_render').value) || 0;

        

        // 加急系数

        const E = parseFloat(document.getElementById('urgentFee').value);


        // 计算总复杂度 ΣCi

        const totalCi = (q_basic * 1.0) + (q_high * 1.5) + (q_immune * 1.2) + (q_render * 1.0);

        

        // 总价计算

        const finalPrice = (B * totalCi) * (1 + E);

        

        // 分段付款计算 [2, 3]

        const depositPrice = finalPrice * 0.4;

        const balancePrice = finalPrice * 0.6;


        // 更新界面

        document.getElementById('totalPrice').innerText = "¥ " + finalPrice.toLocaleString();

        document.getElementById('deposit').innerText = "¥ " + depositPrice.toLocaleString();

        document.getElementById('balance').innerText = "¥ " + balancePrice.toLocaleString();

    }


    // 页面加载时初始化计算一次

    window.onload = calculate;

</script>
